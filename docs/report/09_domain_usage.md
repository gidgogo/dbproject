# 9-부록. 도메인별 사용 컬럼 (자동 생성)

|---|---|---|---|---|
| d_count | smallint | `CHECK ((VALUE >= 0))` | 경기 단위 횟수: 0 이상 정수 (타수, 안타, 삼진 등) | team_game.runs, team_game.hits, team_game.errors, team_game.walks_hbp, inning_score.runs, batter_game.at_bats, batter_game.hits, batter_game.rbi, batter_game.runs, pitcher_game.batters_faced, pitcher_game.at_bats, pitcher_game.pitches, pitcher_game.hits_allowed, pitcher_game.home_runs_allowed, pitcher_game.walks_hbp, pitcher_game.strikeouts, pitcher_game.runs_allowed, pitcher_game.earned_runs |

| d_game_id | character varying(14) | `CHECK (((VALUE)::text ~ '^[0-9]{8}_[A-Z]{4}[0-9]$'::text))` | 경기ID: kbodata 형식 YYYYMMDD_원정팀코드홈팀코드더블헤더 (예: 20250601_OBWO0) | game.game_id, team_game.game_id, inning_score.game_id, batter_game.game_id, plate_appearance.game_id, pitcher_game.game_id, game_event.game_id, game_umpire.game_id, pitch.game_id |

| d_inning | smallint | `CHECK (((VALUE >= 1) AND (VALUE <= 18)))` | 이닝: 1~18 (kbodata가 18회까지 지원) | inning_score.inning, plate_appearance.inning, game_event.inning, pitch.inning |
| d_outs | integer | `CHECK ((VALUE >= 0))` | 아웃카운트: 투수 이닝을 아웃 수로 저장 (6⅓이닝 = 19) | pitcher_game.outs_recorded, season_pitching_official.outs_recorded |
| d_person_name | character varying(30) | `CHECK ((length(btrim((VALUE)::text)) > 0))` | 인명: 공백만으로 된 값 금지 | player.player_name, player_season.name_in_season, game_event.player_name_raw, umpire.umpire_name |
| d_player_id | integer | `CHECK ((VALUE > 0))` | 선수ID: KBO 공식 홈페이지 playerId (예: 76232) | player.player_id, player_season.player_id, player_salary.player_id, batter_game.player_id, plate_appearance.player_id, pitcher_game.player_id, game_event.player_id, pitch.pitcher_id, pitch.batter_id, season_batting_official.player_id, season_pitching_official.player_id |
| d_rate | numeric(4,3) | `CHECK (((VALUE >= (0)::numeric) AND (VALUE <= (1)::numeric)))` | 비율(0~1): 타율, 출루율, 승률 등 소수 셋째 자리 | season_batting_official.avg, season_batting_official.obp, team_rank_official.win_pct |
| d_season | smallint | `CHECK (((VALUE >= 1982) AND (VALUE <= 2100)))` | 시즌: KBO 리그 출범(1982) 이후 연도 | team.first_season, team_season.season, player.draft_year, player.join_year, player_season.season, player_salary.season, game.season, season_batting_official.season, season_pitching_official.season, team_rank_official.season |
| d_slg | numeric(5,3) | `CHECK (((VALUE >= (0)::numeric) AND (VALUE <= (5)::numeric)))` | 장타 비율: 장타율(최대 4.000), OPS(최대 5.000) | season_batting_official.slg, season_batting_official.ops |
| d_team_id | character(2) | `CHECK ((VALUE ~ '^[A-Z]{2}$'::text))` | 팀ID: KBO 팀코드 2자리 영문 대문자 (OB, LT, SS, HH, LG, HT, SK, WO, NC, KT) | team.team_id, team_season.team_id, player_season.team_id, game.away_team_id, game.home_team_id, team_game.team_id, inning_score.team_id, batter_game.team_id, pitcher_game.team_id, game_event.team_id, season_batting_official.team_id, season_pitching_official.team_id, team_rank_official.team_id |
